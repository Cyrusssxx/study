# 每题笔记功能 · 跨项目复用文档

> 来源：`pwa/js/backend.js`（存储接口）、`pwa/quiz.html`（前端编辑器）、`pwa/css/style.css`（样式）
> 适用：需要"按题目 ID 存一条富文本笔记（支持加粗/斜体/标题/4 色高亮/截图贴图）"的任何项目。
> 数据零后端依赖：笔记存在浏览器 **IndexedDB**，通过 PWA 自带的 `/api/note/<question_id>` 接口读写。

---

## 1. 功能概述

- 每题一条笔记，键为 `question_id`。
- 富文本：加粗、斜体、H1/H2 标题、4 色高亮、清除格式。
- 支持 Ctrl+V 直接粘贴截图作为笔记贴图（图片压缩为 dataURL 入库）。
- 空笔记（文字+图片都空）= 删除记录，不占存储。
- 输入停顿即自动保存（防抖 800ms），也可手动点「保存笔记」。
- 装载时对 HTML 做**白名单 sanitize**，防御 XSS。

---

## 2. 存储接口（后端）

接口由 PWA 的 `backend.js` 统一路由处理。下面只列笔记相关部分，移植时需自备：`SUBJECTS`（科目白名单）、`jsonResp()`（JSON 响应封装）、`now()`（时间戳）、`DB_NAME`/`DB_VER`（库名与版本）、请求解析得到的 `seg`(路径段数组)、`body`(解析后的 JSON)、`opts`(含 method)。

### 2.1 路由：`/api/note/<question_id>`（`backend.js` L589–604）

```js
// ---------- 笔记 ----------
if (seg[1] === 'note') {
    const qid = seg[2];
    const subject = qid.split('_')[0];
    if (!SUBJECTS[subject]) return jsonResp({ error: '无效的题目ID' }, 400);
    if ((opts.method || 'GET') === 'GET') {
        const n = await dbGet('notes', qid);
        return jsonResp({ question_id: qid, note: n ? n.content : '', images: (n && n.images) || [] });
    }
    const content = (body.content || '').trim();
    const images = Array.isArray(body.images) ? body.images : [];
    if (content || images.length) {
        await dbPut('notes', { question_id: qid, subject, content, images, updated_at: now() });
    } else {
        await dbDelete('notes', qid);   // 空笔记即删除
    }
    return jsonResp({ success: true, note: content, images });
}
```

**契约**

| 方法 | 入参 | 出参 |
|---|---|---|
| `GET /api/note/<qid>` | — | `{ question_id, note, images }` |
| `POST /api/note/<qid>` | `{ content: string, images: string[] }` | `{ success, note, images }` |

- `subject` 从 `question_id` 前缀按 `_` 切分得到（如 `os_0042` → `os`），用于校验 `SUBJECTS`。
- 有内容 → upsert；`content` 与 `images` 都空 → 删除整条。

### 2.2 IndexedDB 建库（`backend.js` L90–116，节选）

`notes` 与 `wrong`/`favorites` 并列，均以 `question_id` 为 `keyPath`：

```js
function openDB() {
    if (_dbPromise) return _dbPromise;
    _dbPromise = new Promise((resolve, reject) => {
        const req = indexedDB.open(DB_NAME, DB_VER);
        req.onupgradeneeded = () => {
            const db = req.result;
            if (!db.objectStoreNames.contains('progress')) {
                const prog = db.createObjectStore('progress', { keyPath: 'pk', autoIncrement: true });
                prog.createIndex('by_qid', 'question_id');
                prog.createIndex('by_subject', 'subject');
            }
            for (const s of ['wrong', 'favorites', 'notes']) {
                if (!db.objectStoreNames.contains(s)) db.createObjectStore(s, { keyPath: 'question_id' });
            }
            if (!db.objectStoreNames.contains('exams')) {
                db.createObjectStore('exams', { keyPath: 'id', autoIncrement: true });
            }
            if (!db.objectStoreNames.contains('daka_progress')) {
                db.createObjectStore('daka_progress', { keyPath: 'question_id' });
            }
        };
        req.onsuccess = () => resolve(req.result);
        req.onerror = () => reject(req.error);
    });
    return _dbPromise;
}
```

### 2.3 通用 Promise 化读写封装（`backend.js` L119–160）

移植时整段复制即可，`notes` 只是其中一个 store：

```js
async function dbAll(store) {
    const db = await openDB();
    return new Promise((res, rej) => {
        const r = db.transaction(store).objectStore(store).getAll();
        r.onsuccess = () => res(r.result);
        r.onerror = () => rej(r.error);
    });
}
async function dbGet(store, key) {
    const db = await openDB();
    return new Promise((res, rej) => {
        const r = db.transaction(store).objectStore(store).get(key);
        r.onsuccess = () => res(r.result);
        r.onerror = () => rej(r.error);
    });
}
async function dbPut(store, val) {
    const db = await openDB();
    return new Promise((res, rej) => {
        const tx = db.transaction(store, 'readwrite');
        const r = tx.objectStore(store).put(val);
        r.onsuccess = () => res(r.result);
        tx.onerror = () => rej(tx.error);
    });
}
async function dbDelete(store, key) {
    const db = await openDB();
    return new Promise((res, rej) => {
        const tx = db.transaction(store, 'readwrite');
        tx.objectStore(store).delete(key);
        tx.oncomplete = () => res();
        tx.onerror = () => rej(tx.error);
    });
}
function now() {
    const d = new Date(), p = n => String(n).padStart(2, '0');
    return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}`;
}
```

### 2.4 数据模型

`notes` store 中一条记录：

```ts
{
  question_id: string,   // 主键，如 "os_0042"
  subject: string,       // 由 question_id 前缀解析，如 "os"
  content: string,       // 富文本 HTML（白名单内标签）
  images: string[],      // 贴图 dataURL 数组
  updated_at: string     // "YYYY-MM-DD HH:MM:SS"
}
```

### 3.2 保存（`quiz.html` L887–918）

```js
async function saveNote(isAuto = false) {
    const q = questions[currentIndex];           // 自行替换为当前题对象
    const inputEl = document.getElementById('noteInput');
    const content = inputEl ? serializeNote(inputEl) : (q.note || '');
    const hadNote = !!(q.note || (q.note_images || []).length);
    try {
        const resp = await api(`/api/note/${q.id}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ content, images: q.note_images || [] })
        });
        const result = await resp.json();
        q.note = result.note || '';
        q.note_images = result.images || [];
        const hasNote = !!(q.note || (q.note_images || []).length);
        const hint = document.getElementById('noteHint');
        hint.textContent = hasNote ? (isAuto ? '已自动保存 ✓' : '已保存 ✓') : (hadNote ? '已删除笔记' : '笔记是空的，没有保存');
        setTimeout(() => { if (hint) hint.textContent = ''; }, 2000);
        const btn = document.getElementById('noteSaveBtn');
        if (btn) btn.classList.toggle('saved', hasNote);
    } catch (e) {
        if (isAuto) {
            const hint = document.getElementById('noteHint');
            if (hint) hint.textContent = '自动保存失败，请点「保存笔记」重试';
        } else {
            alert('笔记保存失败: ' + e.message);
        }
    }
}
```

### 3.3 装载 + 白名单 sanitize（防 XSS，`quiz.html` L925–994）

```js
const NOTE_ALLOWED_TAGS = new Set(['B','I','U','STRONG','EM','H1','H2','H3','P','BR','DIV','SPAN','UL','OL','LI','BLOCKQUOTE']);
const NOTE_ALLOWED_CLASS_RE = /^(hl-|note-hl-)?(yellow|green|blue|pink)$/;

function sanitizeNoteHtml(html) {
    if (!html) return '';
    if (html.indexOf('<') < 0) return '';          // 纯文本极速判定
    const t = document.createElement('template');
    t.innerHTML = html;
    (function walk(node) {
        const children = [...node.childNodes];
        for (const c of children) {
            if (c.nodeType === 1) {
                if (!NOTE_ALLOWED_TAGS.has(c.tagName)) {
                    while (c.firstChild) node.insertBefore(c.firstChild, c);
                    node.removeChild(c);
                } else {
                    for (const a of [...c.attributes]) {
                        if (a.name === 'style') {
                            const ok = a.value.split(';').map(s => s.trim()).filter(s => {
                                const m = s.match(/^([a-z-]+):/);
                                return m && (m[1] === 'color' || m[1] === 'background-color' || m[1] === 'background');
                            });
                            if (ok.length) c.setAttribute('style', ok.join('; '));
                            else c.removeAttribute('style');
                        } else if (a.name === 'class') {
                            if (!NOTE_ALLOWED_CLASS_RE.test(a.value)) c.removeAttribute('class');
                        } else {
                            c.removeAttribute(a.name);
                        }
                    }
                    walk(c);
                }
            } else if (c.nodeType === 8) {
                c.remove();                          // 丢弃注释节点
            }
        }
    })(t.content);
    return t.innerHTML;
}
function isHtmlNote(s) {
    return /<(b|i|u|strong|em|h[1-3]|p|br|div|span|ul|ol|li|blockquote)\b/i.test(s);
}
function loadNoteIntoEditor(el, raw) {
    if (!el) return;
    el.innerHTML = '';
    if (!raw) { updateNotePlaceholder(el); return; }
    if (isHtmlNote(raw)) {
        el.innerHTML = sanitizeNoteHtml(raw);
    } else {
        // 旧纯文本：转义后换行转 <br>，再整体白名单化
        el.innerHTML = sanitizeNoteHtml(raw.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/\n/g, '<br>'));
    }
    updateNotePlaceholder(el);
}
function serializeNote(el) {
    if (!el) return '';
    const text = el.textContent.replace(/ /g, ' ').trim();
    if (!text) return '';                            // 空内容（仅 <br> 等）存为空串
    return el.innerHTML;
}
function updateNotePlaceholder(el) {
    if (!el) return;
    const text = el.textContent.trim();
    el.classList.toggle('is-empty', !text);
}
```

### 3.4 工具条绑定 + 高亮 + 清除格式（`quiz.html` L995–1077）

```js
function bindNoteToolbar(el) {
    if (!el) return;
    const panel = el.closest('.note-panel');
    if (!panel) return;
    const tb = panel.querySelector('.note-toolbar');
    if (!tb || tb.dataset.bound) return;
    tb.dataset.bound = '1';
    tb.addEventListener('mousedown', e => e.preventDefault());   // 按下不丢选区
    tb.addEventListener('click', e => {
        const btn = e.target.closest('[data-cmd]');
        if (!btn) return;
        el.focus();
        const cmd = btn.dataset.cmd;
        if (cmd === 'bold') document.execCommand('bold');
        else if (cmd === 'italic') document.execCommand('italic');
        else if (cmd === 'h1') document.execCommand('formatBlock', false, 'H1');
        else if (cmd === 'h2') document.execCommand('formatBlock', false, 'H2');
        else if (cmd === 'hl') applyNoteHighlight(el, btn.dataset.color);
        else if (cmd === 'clear') clearNoteFormat(el);
        onNoteInput();
    });
    el.addEventListener('keydown', e => {
        if (!(e.ctrlKey || e.metaKey)) return;
        if (e.key === 'b' || e.key === 'B') { e.preventDefault(); document.execCommand('bold'); onNoteInput(); }
        else if (e.key === 'i' || e.key === 'I') { e.preventDefault(); document.execCommand('italic'); onNoteInput(); }
    });
    el.addEventListener('input', () => updateNotePlaceholder(el));
    el.addEventListener('blur', () => updateNotePlaceholder(el));
}
function applyNoteHighlight(el, color) {
    const sel = window.getSelection();
    if (!sel || sel.isCollapsed || !sel.rangeCount) return;
    const range = sel.getRangeAt(0);
    if (!el.contains(range.startContainer) || !el.contains(range.endContainer)) return;
    const ok = document.execCommand('hiliteColor', false, ({
        yellow: '#fff3a3', green: '#b8e6c0', blue: '#b3d9f2', pink: '#f5c2d5'
    })[color]);
    if (ok) return;
    try {                                           // 后备：手动包 span
        const span = document.createElement('span');
        span.className = 'hl hl-' + color;
        span.appendChild(range.extractContents());
        range.insertNode(span);
    } catch (e) { /* 选区越界，忽略 */ }
}
// 「清除格式」三步：行内(粗体/斜体/颜色) + 块级(H1/H2/H3/引用) + 高亮
// 只调 removeFormat 清不掉 formatBlock 产生的 H1/H2，也剥不掉高亮 span/内联 background
function clearNoteFormat(el) {
    el.focus();
    const sel = window.getSelection();
    if (!sel || sel.rangeCount === 0) return onNoteInput();
    document.execCommand('removeFormat');                       // 1) 行内
    try { document.execCommand('formatBlock', false, 'P'); } catch (e) {}  // 2) 块级 → P
    stripNoteHighlight(sel.getRangeAt(0), el);                  // 3) 高亮
    onNoteInput();
}
function stripNoteHighlight(range, root) {
    root.querySelectorAll('*').forEach(s => {
        if (!range.intersectsNode(s)) return;
        if (s.style && (s.style.backgroundColor || s.style.background)) {
            s.style.backgroundColor = '';
            s.style.background = '';
            if (s.getAttribute('style') === '') s.removeAttribute('style');
        }
        if (s.classList && s.classList.contains('hl')) {
            const p = s.parentNode;
            while (s.firstChild) p.insertBefore(s.firstChild, s);
            p.removeChild(s);
        }
    });
}
```

> `onNoteEdit` 用于展开面板后首次聚焦；`autoResizeNote(el)` 按内容高度自适应 textarea/div 高度，按需实现。

## 6. 依赖与注意点

- **`contenteditable` + `execCommand`**：`bold`/`italic`/`formatBlock`/`hiliteColor`/`removeFormat` 虽被标记 deprecated，但主流浏览器仍广泛支持，零依赖。
- **白名单 sanitize 必经**：`loadNoteIntoEditor` 必须先过 `sanitizeNoteHtml` 再 `innerHTML`，否则存进 IndexedDB 的富文本会在装载时造成 XSS。
- **空即删**：`content` 与 `images` 都空时 POST 走 `dbDelete`，不残留空记录。
- **`serializeNote` 空气判断**：仅含 `<br>`/空标签时 `textContent.trim()` 为空，返回空串，避免"空气笔记"被存成非空。
- **`clearNoteFormat` 三步缺一不可**：`removeFormat` 只清行内，清不掉 `formatBlock` 的 H1/H2，也剥不掉高亮 span/内联 background——三类必须分别处理，否则"清除格式不生效"。
- **`subject` 校验**：`question_id` 必须含 `_`，前缀在 `SUBJECTS` 内，否则 400。新项目若无科目概念，可把该校验改为始终通过。
